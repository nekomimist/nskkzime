cmake_minimum_required(VERSION 3.24)
project(nskkzime VERSION 0.0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

add_library(project_options INTERFACE)
if(MSVC)
  target_compile_options(project_options INTERFACE /W4 /WX /utf-8 /permissive- /EHsc)
  target_compile_definitions(project_options INTERFACE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_compile_options(project_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()

add_library(skk_core STATIC src/core/romaji.cpp src/core/engine.cpp src/core/text.cpp)
target_include_directories(skk_core PUBLIC include)
target_link_libraries(skk_core PUBLIC project_options)
add_library(skk_dictionary STATIC src/dictionary/skk_dictionary.cpp src/dictionary/learning.cpp src/dictionary/user_export.cpp)
target_link_libraries(skk_dictionary PUBLIC skk_core)
add_executable(skk_trace src/tools/trace.cpp)
target_link_libraries(skk_trace PRIVATE skk_dictionary)

if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE skk_core)
  add_test(NAME core COMMAND core_tests)
  add_executable(registration_tests tests/registration_tests.cpp)
  target_link_libraries(registration_tests PRIVATE skk_core)
  add_test(NAME registration COMMAND registration_tests)
  add_executable(dictionary_tests tests/dictionary_tests.cpp)
  target_link_libraries(dictionary_tests PRIVATE skk_dictionary)
  add_test(NAME dictionary COMMAND dictionary_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/basic.utf8.skk)
  add_test(NAME dictionary_trace COMMAND skk_trace --dict ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/basic.utf8.skk "Jisho<Space><Enter>")
  set_tests_properties(dictionary_trace PROPERTIES PASS_REGULAR_EXPRESSION "辞書")
  add_test(NAME dictionary_missing_file COMMAND skk_trace --dict ${CMAKE_CURRENT_BINARY_DIR}/missing-dictionary.skk "Kanji<Space>")
  set_tests_properties(dictionary_missing_file PROPERTIES WILL_FAIL TRUE)
endif()

if(WIN32)
  add_library(nskkzime_runtime STATIC src/windows/dictionary_settings.cpp src/windows/runtime_dictionary.cpp
    src/windows/learning_ipc.cpp src/windows/learning_client.cpp)
  target_include_directories(nskkzime_runtime PUBLIC src/windows)
  target_link_libraries(nskkzime_runtime PUBLIC skk_dictionary shell32 ole32 uuid advapi32 bcrypt)
  add_executable(nskkzime_dictionary_server src/tools/dictionary_server.cpp)
  target_link_libraries(nskkzime_dictionary_server PRIVATE nskkzime_runtime)
  add_executable(nskkzime_config src/tools/config.cpp)
  target_link_libraries(nskkzime_config PRIVATE nskkzime_runtime)
  add_executable(nskkzime_user_dictionary src/tools/user_dictionary.cpp)
  target_link_libraries(nskkzime_user_dictionary PRIVATE nskkzime_runtime)
  add_dependencies(nskkzime_user_dictionary nskkzime_dictionary_server)
  add_executable(ime_probe WIN32 src/tools/ime_probe.cpp)
  target_include_directories(ime_probe PRIVATE src/windows)
  target_link_libraries(ime_probe PRIVATE project_options ole32 oleaut32 uuid user32 gdi32)
  add_library(nskkzime_tsf SHARED src/windows/text_service.cpp src/windows/module.cpp
    src/windows/registration.cpp src/windows/display_attributes.cpp
    src/windows/candidate_ui.cpp src/windows/input_mode.cpp src/windows/context_text.cpp src/windows/nskkzime.def)
  target_include_directories(nskkzime_tsf PRIVATE src/windows)
  target_link_libraries(nskkzime_tsf PRIVATE nskkzime_runtime ole32 oleaut32 advapi32 uuid user32 gdi32)
  add_dependencies(nskkzime_tsf nskkzime_dictionary_server)
  # Same native controls/TSF profile, but activate this build's COM DLL only
  # inside the probe. Normal applications continue to use the registered DLL.
  add_executable(ime_probe_local WIN32 src/tools/ime_probe.cpp src/tools/ime_probe_local.manifest)
  target_include_directories(ime_probe_local PRIVATE src/windows)
  target_link_libraries(ime_probe_local PRIVATE project_options ole32 oleaut32 uuid user32 gdi32)
  target_compile_definitions(ime_probe_local PRIVATE NSKKZIME_LOCAL_PROBE)
  add_dependencies(ime_probe_local nskkzime_tsf)
  if(BUILD_TESTING)
    add_executable(ime_probe_isolated WIN32 src/tools/ime_probe.cpp src/tools/ime_probe_local.manifest)
    target_include_directories(ime_probe_isolated PRIVATE src/windows)
    target_link_libraries(ime_probe_isolated PRIVATE project_options ole32 oleaut32 uuid user32 gdi32)
    target_compile_definitions(ime_probe_isolated PRIVATE NSKKZIME_PROBE_DICTIONARY=L"${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/probe.utf8.skk")
    add_dependencies(ime_probe_isolated nskkzime_tsf)
    add_executable(learning_tests tests/learning_tests.cpp)
    target_link_libraries(learning_tests PRIVATE nskkzime_runtime)
    add_test(NAME learning COMMAND learning_tests $<TARGET_FILE:nskkzime_dictionary_server>)
    set_tests_properties(learning PROPERTIES TIMEOUT 60)
    add_executable(user_dictionary_tests tests/user_dictionary_tests.cpp)
    target_link_libraries(user_dictionary_tests PRIVATE nskkzime_runtime)
    add_test(NAME user_dictionary COMMAND user_dictionary_tests)
    add_test(NAME user_dictionary_cli COMMAND ${CMAKE_COMMAND}
      -DUSER_EXE=$<TARGET_FILE:nskkzime_user_dictionary>
      -DTEST_EXE=$<TARGET_FILE:user_dictionary_tests>
      -DTEST_ROOT=${CMAKE_CURRENT_BINARY_DIR}/user-cli-$<CONFIG>
      -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/user_dictionary_cli.cmake)
    set_tests_properties(user_dictionary_cli PROPERTIES TIMEOUT 40)
    add_executable(tsf_learning tests/tsf_learning.cpp)
    target_link_libraries(tsf_learning PRIVATE nskkzime_runtime oleaut32 user32)
    add_dependencies(tsf_learning nskkzime_tsf)
    add_test(NAME tsf_learning COMMAND tsf_learning $<TARGET_FILE:nskkzime_tsf> ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/basic.utf8.skk)
    set_tests_properties(tsf_learning PROPERTIES TIMEOUT 40)
    add_executable(settings_tests tests/settings_tests.cpp)
    target_link_libraries(settings_tests PRIVATE nskkzime_runtime)
    add_test(NAME settings COMMAND settings_tests)
    add_test(NAME config_cli COMMAND ${CMAKE_COMMAND}
      -DCONFIG_EXE=$<TARGET_FILE:nskkzime_config>
      -DDICTIONARY=${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/basic.utf8.skk
      -DTEST_ROOT=${CMAKE_CURRENT_BINARY_DIR}/config-cli-$<CONFIG>
      -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/config_cli.cmake)
    add_executable(candidate_ui_tests tests/candidate_ui_tests.cpp src/windows/candidate_ui.cpp)
    target_include_directories(candidate_ui_tests PRIVATE src/windows)
    target_link_libraries(candidate_ui_tests PRIVATE skk_core ole32 oleaut32 uuid user32 gdi32)
    add_test(NAME candidate_ui COMMAND candidate_ui_tests)
    # Link the same adapter sources to exercise deferred click requests without
    # injecting native input or relying on a registered IME/interactive desktop.
    add_executable(tsf_candidate_click tests/tsf_candidate_click.cpp src/windows/text_service.cpp
      src/windows/candidate_ui.cpp src/windows/input_mode.cpp src/windows/context_text.cpp src/windows/display_attributes.cpp)
    target_include_directories(tsf_candidate_click PRIVATE src/windows)
    target_link_libraries(tsf_candidate_click PRIVATE nskkzime_runtime ole32 oleaut32 uuid user32 gdi32)
    add_dependencies(tsf_candidate_click nskkzime_dictionary_server)
    add_test(NAME tsf_candidate_click COMMAND tsf_candidate_click ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/basic.utf8.skk)
    set_tests_properties(tsf_candidate_click PROPERTIES TIMEOUT 40)
    add_executable(input_mode_tests tests/input_mode_tests.cpp src/windows/text_service.cpp
      src/windows/candidate_ui.cpp src/windows/input_mode.cpp src/windows/context_text.cpp src/windows/display_attributes.cpp)
    target_link_libraries(input_mode_tests PRIVATE nskkzime_runtime ole32 oleaut32 uuid user32 gdi32)
    add_dependencies(input_mode_tests nskkzime_dictionary_server)
    add_test(NAME input_mode COMMAND input_mode_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/basic.utf8.skk)
    set_tests_properties(input_mode PROPERTIES TIMEOUT 40)
    add_executable(context_text_tests tests/context_text_tests.cpp src/windows/text_service.cpp
      src/windows/candidate_ui.cpp src/windows/input_mode.cpp src/windows/context_text.cpp src/windows/display_attributes.cpp)
    target_link_libraries(context_text_tests PRIVATE nskkzime_runtime ole32 oleaut32 uuid user32 gdi32)
    add_test(NAME context_text COMMAND context_text_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/basic.utf8.skk)
    add_executable(tsf_local_activation tests/tsf_local_activation.cpp src/tools/ime_probe_local.manifest)
    target_include_directories(tsf_local_activation PRIVATE src/windows)
    target_link_libraries(tsf_local_activation PRIVATE project_options ole32 uuid)
    add_dependencies(tsf_local_activation nskkzime_tsf)
    add_test(NAME tsf_local_activation COMMAND tsf_local_activation)
    add_executable(tsf_smoke tests/tsf_smoke.cpp)
    target_include_directories(tsf_smoke PRIVATE src/windows)
    target_link_libraries(tsf_smoke PRIVATE project_options ole32 oleaut32 uuid user32)
    target_compile_definitions(tsf_smoke PRIVATE NSKKZIME_TEST_DICTIONARY=L"${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/basic.utf8.skk")
    add_test(NAME tsf_smoke COMMAND tsf_smoke $<TARGET_FILE:nskkzime_tsf>)
    add_test(NAME tsf_dictionary COMMAND tsf_smoke $<TARGET_FILE:nskkzime_tsf> ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/basic.utf8.skk)
    add_test(NAME tsf_missing_dictionary COMMAND tsf_smoke $<TARGET_FILE:nskkzime_tsf> ${CMAKE_CURRENT_BINARY_DIR}/missing-dictionary.skk --expect-empty)
    file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/test-settings.ini CONTENT "version=1\ndictionary=${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/basic.utf8.skk\n")
    file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/invalid-settings.ini CONTENT "version=unsupported\n")
    add_test(NAME tsf_settings COMMAND tsf_smoke $<TARGET_FILE:nskkzime_tsf> ${CMAKE_CURRENT_BINARY_DIR}/test-settings.ini --config)
    add_test(NAME tsf_invalid_settings COMMAND tsf_smoke $<TARGET_FILE:nskkzime_tsf> ${CMAKE_CURRENT_BINARY_DIR}/invalid-settings.ini --config-empty)
  endif()
endif()
